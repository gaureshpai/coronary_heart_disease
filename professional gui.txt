import pandas as pd

# ML imports
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import StandardScaler
from sklearn.model_selection import train_test_split
from sklearn.metrics import accuracy_score

from sklearn.neighbors import KNeighborsClassifier
from sklearn.linear_model import LogisticRegression
from sklearn.naive_bayes import GaussianNB
from sklearn.tree import DecisionTreeClassifier
from sklearn.svm import SVC
from sklearn.ensemble import RandomForestClassifier

# GUI imports
import tkinter as tk

# ---------- Data loading & preprocessing ----------
DATA_PATH = r"C:\Users\HP\Desktop\project\framingham.csv"   # <-- update if needed

df = pd.read_csv(DATA_PATH)

# drop columns and outliers as in your original code
df1 = df.drop(columns=["education"])
features_to_drop = ['currentSmoker', 'diaBP']
df2 = df1.drop(columns=features_to_drop)

imputer = SimpleImputer(strategy='most_frequent')
df3 = pd.DataFrame(imputer.fit_transform(df2), columns=df2.columns, index=df2.index)

# remove extreme outliers as in original
df3 = df3[~(df3['sysBP'] > 220)]
df3 = df3[~(df3['BMI'] > 43)]
df3 = df3[~(df3['heartRate'] > 125)]
df3 = df3[~(df3['glucose'] > 200)]
df3 = df3[~(df3['totChol'] > 450)]

# Standardise selected features
scaler = StandardScaler()
cols_to_standardise = ['age','cigsPerDay','totChol','sysBP','BMI','heartRate','glucose']
df3[cols_to_standardise] = scaler.fit_transform(df3[cols_to_standardise])

# feature / label split
X = df3.drop(columns=["TenYearCHD"])
y = df3["TenYearCHD"]

# Preserve feature order for prediction
feature_order = list(X.columns)

# train/test split
X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2, random_state=40)

# ---------- Train models once and store accuracies ----------
models = {}
accuracies = {}

def train_all_models():
    # KNN
    knn = KNeighborsClassifier(n_neighbors=7)
    knn.fit(X_train, y_train)
    models['KNN'] = knn
    accuracies['KNN'] = round(accuracy_score(y_test, knn.predict(X_test))*100,2)

    # Logistic Regression
    lg = LogisticRegression(max_iter=1000)
    lg.fit(X_train, y_train)
    models['Logistic Regression'] = lg
    accuracies['Logistic Regression'] = round(accuracy_score(y_test, lg.predict(X_test))*100,2)

    # Naive Bayes
    nb = GaussianNB()
    nb.fit(X_train, y_train)
    models['Naive Bayes'] = nb
    accuracies['Naive Bayes'] = round(accuracy_score(y_test, nb.predict(X_test))*100,2)

    # Decision Tree
    dt = DecisionTreeClassifier(min_samples_split=50, random_state=0)
    dt.fit(X_train, y_train)
    models['Decision Tree'] = dt
    accuracies['Decision Tree'] = round(accuracy_score(y_test, dt.predict(X_test))*100,2)

    # SVM
    svc = SVC(C=1, kernel='rbf', probability=True)
    svc.fit(X_train, y_train)
    models['SVM'] = svc
    accuracies['SVM'] = round(accuracy_score(y_test, svc.predict(X_test))*100,2)

    # Random Forest
    rf = RandomForestClassifier(n_estimators=100, random_state=0)
    rf.fit(X_train, y_train)
    models['Random Forest'] = rf
    accuracies['Random Forest'] = round(accuracy_score(y_test, rf.predict(X_test))*100,2)

train_all_models()

# ---------- Helper utilities ----------
def float_or_raise(value):
    try:
        return float(value)
    except:
        raise ValueError("Invalid numeric value: " + str(value))

def build_sample_from_inputs(inputs):
    """
    inputs: dict mapping feature name -> user-supplied value
    Returns: 1xN array with same column order as X
    """
    # We must apply the same scaling to the numeric fields used in scaling
    # First, create a single-row dataframe with raw (unscaled) values for columns that were scaled
    row = {}
    for col in feature_order:
        if col in ['age','cigsPerDay','totChol','sysBP','BMI','heartRate','glucose']:
            row[col] = float_or_raise(inputs.get(col))
        else:
            # the other features are expected as numeric/binary, use float conversion
            row[col] = float_or_raise(inputs.get(col))
    raw_df = pd.DataFrame([row])
    # apply the same scaler transform to the numeric columns
    raw_df[cols_to_standardise] = scaler.transform(raw_df[cols_to_standardise])
    # ensure column order
    sample = raw_df[feature_order].values
    return sample

def get_best_model_name():
    # name of model with highest accuracy
    return max(accuracies.items(), key=lambda kv: kv[1])[0]

    def predict_best(self):
        model_name = get_best_model_name()
        self._predict_with_model_name(model_name)

    def predict_choose_model(self):
        model_name = self.model_choice.get()
        self._predict_with_model_name(model_name)

    def _predict_with_model_name(self, model_name):
        try:
            # gather inputs into dict matching column names
            inputs = {}
            # The feature names expected by model (your original X columns) likely include
            # 'male','age','cigsPerDay','BPMeds','prevalentStroke','prevalentHyp','diabetes',
            # 'totChol','sysBP','BMI','heartRate','glucose'
            for col in feature_order:
                # If the user input field exists, use it; else default 0
                if col in self.inputs:
                    inputs[col] = self.inputs[col].get()
                else:
                    # fallback: if not in GUI, assume 0
                    inputs[col] = "0"
            sample = build_sample_from_inputs(inputs)
            model = models[model_name]
            pred = model.predict(sample)[0]
            # show probability if available
            prob_text = ""
            if hasattr(model, "predict_proba"):
                proba = model.predict_proba(sample)[0]
                # probability of class 1 (positive)
                prob_text = f"  |  Probability (CHD=1): {proba[1]*100:.2f}%"
            if pred == 1:
                self.result_var.set("Prediction: HIGH RISK of CHD")
                self.result_var_label_color("red")
            else:
                self.result_var.set("Prediction: LOW RISK of CHD")
                self.result_var_label_color("green")
            self.conf_var.set(f"Model used: {model_name} (Accuracy: {accuracies[model_name]}%)" + prob_text)
            self.status_var.set("Prediction completed.")
        except Exception as e:
            messagebox.showerror("Error", f"Failed to predict: {e}")
            self.status_var.set("Error during prediction.")

    def clear_fields(self):
        for var in self.inputs.values():
            var.set("")

        names = list(accuracies.keys())
        vals = [accuracies[n] for n in names]
        fig, ax = plt.subplots(figsize=(8,4))
        bars = ax.bar(names, vals)
        ax.set_ylim(0,100)
        ax.set_ylabel("Accuracy (%)")
        ax.set_title("Model comparison (accuracy)")

            ax.text(bar.get_x() + bar.get_width()/2, v + 1.5, f"{v}%", ha='center')
        plt.xticks(rotation=30, ha='right')
        plt.tight_layout()
        plt.show()

    def export_accuracies(self):
